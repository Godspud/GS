class Animal {
    def __init__(self) {
        self.name = "creature";
        self.legs = 4;
    }

    def speak(self) {
        print(f"{self.name} says something");
    }

    def str describe(self) {
        return f"{self.name} has {self.legs} legs";
    }
}

// class (Parent)Child -- parent name goes in parens, before the child's name.
class (Animal)Dog {
    def __init__(self) {
        Animal.__init__(self);
    }

    // Overriding speak(): Dog's own version is used instead of Animal's.
    def speak(self) {
        print(f"{self.name} says Woof!");
    }

    def bark_times(self, int times) {
        for (i in range(times)) {
            print(f"Woof #{i}");
        }
    }
}

Dog d = Dog();
d.name = "Rex";
d.legs = 4;
d.speak();          // Dog's override runs.
d.bark_times(3);
print(d.describe()); // Not overridden -> falls back to Animal's describe().

def summarize(str label, *rest, **opts) {
    print(label);
    print("rest count:");
    print(len(rest));
    int i = 0;
    while (i < len(rest)) {
        print(rest[i]);
        i = i + 1;
    }
    if ("verbose" in opts) {
        print("verbose:");
        print(opts["verbose"]);
    }
    if ("tag" in opts) {
        print("tag:");
        print(opts["tag"]);
    }
}

class Animal {
    def __init__(self, str name) {
        self.name = name;
    }
    def speak(self) {
        return "...";
    }
}

class (Animal)Dog {
    def speak(self) {
        return "Woof";
    }
}

class (Animal)Cat {
    def speak(self) {
        return "Meow";
    }
}

class (Dog)Puppy {
}

Dog d = Dog("Rex");
Cat c = Cat("Tom");
Puppy p = Puppy("Spot");

// statically-typed class values
print(isinstance(d, Dog));      // True
print(isinstance(d, Animal));   // True (base class)
print(isinstance(d, Cat));      // False (unrelated)
print(isinstance(p, Dog));      // True (Puppy extends Dog)
print(isinstance(p, Animal));   // True (grandparent)
print(isinstance(d, Puppy));    // False (d isn't a Puppy, wrong direction)

// any-typed values holding objects
any a1 = Dog("AnyRex");
any a2 = Cat("AnyTom");
print(isinstance(a1, Dog));     // True
print(isinstance(a1, Animal));  // True
print(isinstance(a1, Cat));     // False
print(isinstance(a2, Animal));  // True
print(isinstance(a2, Dog));     // False

// any-typed value NOT holding an object at all
any a3 = 42;
print(isinstance(a3, Dog));     // False, no crash

// statically non-object types -- always False, no error
int n = 5;
str s = "hi";
print(isinstance(n, Dog));      // False
print(isinstance(s, Animal));   // False

// isinstance used in a real conditional
def str describe(any x) {
    if (isinstance(x, Dog)) {
        return "it's a dog";
    } elif (isinstance(x, Animal)) {
        return "it's some animal";
    } else {
        return "not an animal";
    }
}
print(describe(a1));
print(describe(a2));
print(describe(a3));
class MathUtils {
    @staticmethod
    def int square(int x) {
        return x * x;
    }

    @staticmethod
    def int add(int a, int b) {
        return a + b;
    }

    def int instance_method(self, int x) {
        return x + 1;
    }
}

// called via ClassName.method(args) -- the unbound style
print(MathUtils.square(5));
print(MathUtils.add(3, 4));

// called via instance.method(args) -- same static method, no receiver passed
MathUtils m = MathUtils();
print(m.square(6));
print(m.add(10, 20));

// mixing static and instance methods on the same instance
print(m.instance_method(41));

class (MathUtils)ExtendedMath {
    @staticmethod
    def int cube(int x) {
        return x * x * x;
    }
}

// inherited static method, called both ways
print(ExtendedMath.square(7));
ExtendedMath e = ExtendedMath();
print(e.square(8));
print(e.cube(3));
print(ExtendedMath.cube(4));

// static method through 'any'-typed dynamic dispatch
any a = MathUtils();
print(a.square(9));
print(a.add(1, 2));

any ext = ExtendedMath();
print(ext.cube(5));
print(ext.square(10));

// side-effect preservation: calling a static method off a side-effecting
// expression should still run that expression, even though its result
// isn't passed anywhere
int counter = 0;
def MathUtils bump() {
    counter = counter + 1;
    return MathUtils();
}
print(bump().square(2));
print(counter);
